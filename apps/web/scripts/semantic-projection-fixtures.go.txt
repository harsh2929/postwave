type fixtureCase struct {
	Name               string            `json:"name"`
	Note               string            `json:"note"`
	Locale             string            `json:"locale"`
	Slots              map[string]string `json:"slots"`
	Input              json.RawMessage   `json:"input"`
	Normalized         json.RawMessage   `json:"normalized"`
	ValidateDiagnostic Diagnostics       `json:"validate_diagnostics"`
	CompileDiagnostic  Diagnostics       `json:"compile_diagnostics"`
	CompileFailed      bool              `json:"compile_failed"`
	HTML               string            `json:"html"`
	Text               string            `json:"text"`
}

type fixtureFile struct {
	Generator       string        `json:"generator"`
	CompilerVersion string        `json:"compiler_version"`
	HTMLVersion     string        `json:"html_version"`
	TextVersion     string        `json:"text_version"`
	Cases           []fixtureCase `json:"cases"`
}

// TestExportWebProjectionFixtures is a generator, not an assertion. It only
// runs when POSTWAVE_WEB_FIXTURE_OUT is set.
func TestExportWebProjectionFixtures(t *testing.T) {
	out := os.Getenv("POSTWAVE_WEB_FIXTURE_OUT")
	if out == "" {
		t.Skip("POSTWAVE_WEB_FIXTURE_OUT is unset")
	}
	file := fixtureFile{
		Generator:       "apps/web/scripts/semantic-projection-fixtures.go.txt",
		CompilerVersion: CompilerVersion,
		HTMLVersion:     HTMLRendererVersion,
		TextVersion:     TextRendererVersion,
	}
	for _, corpus := range fixtureCorpus() {
		var input DocumentV2
		if err := json.Unmarshal(corpus.Input, &input); err != nil {
			t.Fatalf("%s: corpus JSON is not a semantic document: %v", corpus.Name, err)
		}
		normalized, diagnostics := NormalizeAndValidateV2(input)
		normalizedJSON, err := json.Marshal(normalized)
		if err != nil {
			t.Fatalf("%s: normalized document does not marshal: %v", corpus.Name, err)
		}
		corpus.Normalized = normalizedJSON
		corpus.ValidateDiagnostic = nonNilDiagnostics(diagnostics)

		slots := map[SlotID]string{}
		for key, value := range corpus.Slots {
			slots[SlotID(key)] = value
		}
		// The zero StructuredPolicy is deliberate. A draft editor holds no
		// sender-eligibility evidence, so provider markup is never emitted and
		// the preview shows exactly the projection the reader would get.
		result, compileDiagnostics := CompileV2(input, CompileOptions{Locale: corpus.Locale, Slots: slots})
		corpus.CompileDiagnostic = nonNilDiagnostics(compileDiagnostics)
		corpus.CompileFailed = compileDiagnostics.HasErrors()
		corpus.HTML = result.HTML
		corpus.Text = result.Text
		file.Cases = append(file.Cases, corpus)
	}
	encoded, err := json.MarshalIndent(file, "", "  ")
	if err != nil {
		t.Fatalf("marshal fixture file: %v", err)
	}
	encoded = append(encoded, '\n')
	if err := os.WriteFile(out, encoded, 0o644); err != nil {
		t.Fatalf("write fixture file: %v", err)
	}
	t.Logf("wrote %d cases to %s", len(file.Cases), out)
}

func nonNilDiagnostics(in Diagnostics) Diagnostics {
	if in == nil {
		return Diagnostics{}
	}
	return in
}

// roleMatrixCase pins the ENTIRE role/type table rather than one example of a
// mismatch.
//
// Role is not decoration: it is what lets action extraction tell a primary CTA
// from an unsubscribe link, and what makes an image informative or decorative.
// A mirror of the table that silently gained an extra pair would let the editor
// author something the server rejects, and a single hand-picked mismatch case
// would not notice. So every (type, role) pair in the closed sets gets a node,
// and Go decides which of the 450 are wrong.
func roleMatrixCase() fixtureCase {
	types := []NodeType{
		NodeHeading, NodeParagraph, NodeList, NodeAnswer, NodeLegal, NodeAction,
		NodeImage, NodeDateTime, NodeMoney, NodeDeadline, NodeOffer, NodeLocation,
		NodeProduct, NodeFact, NodePersonalization, NodeStructuredData,
		NodeDivider, NodePoll,
	}
	roles := []NodeRole{
		RoleHeading, RoleBody, RoleAnswer, RoleLegal, RoleFooter, RolePrimaryAction,
		RoleSecondaryAction, RoleUnsubscribe, RolePreferences, RoleViewInBrowser,
		RoleSocial, RoleContact, RoleInformativeImage, RoleDecorativeImage,
		RoleEventDateTime, RolePrice, RoleDeadline, RoleOffer, RoleLocation,
		RoleProduct, RoleProductFact, RolePersonalizationSlot, RoleStructuredData,
		RoleDivider, RolePoll,
	}
	nodes := make([]Node, 0, len(types)*len(roles))
	index := 0
	for _, nodeType := range types {
		for _, role := range roles {
			id := NodeID(fmt.Sprintf("n%03d", index))
			nodes = append(nodes, Node{
				ID: id, Type: nodeType, Role: role, Version: NodeVersionV2,
				Conditions: []Condition{}, Localizations: map[string]NodeContent{},
				Provenance: Provenance{Source: SourceUser},
				Content:    matrixContent(nodeType, index),
			})
			index++
		}
	}
	encoded, err := json.Marshal(DocumentV2{
		SchemaVersion: SchemaVersionV2, Language: "en-US", Direction: DirectionLTR,
		Nodes: nodes,
	})
	if err != nil {
		panic(err)
	}
	return fixtureCase{
		Name:  "role_matrix",
		Note:  "Every (node type, role) pair in the closed sets. Go decides which pairs are valid; the mirror must agree on all of them.",
		Input: encoded,
	}
}

func matrixContent(nodeType NodeType, index int) NodeContent {
	text := RichText{Runs: []InlineRun{{Kind: InlineText, Text: "x"}}}
	when := &DateTimeContent{DisplayValue: "now", RFC3339: "2026-08-14T16:00:00+05:30", Timezone: "Asia/Kolkata"}
	switch nodeType {
	case NodeHeading:
		return NodeContent{Heading: &HeadingContent{Level: 2, Text: text}}
	case NodeParagraph:
		return NodeContent{Paragraph: &ParagraphContent{Text: text}}
	case NodeList:
		return NodeContent{List: &ListContent{Items: []RichText{text}}}
	case NodeAnswer:
		return NodeContent{Answer: &AnswerContent{Question: text, Answer: text}}
	case NodeLegal:
		return NodeContent{Legal: &LegalContent{Text: text}}
	case NodeAction:
		return NodeContent{Action: &ActionContent{
			ActionID: ActionID(fmt.Sprintf("act%03d", index)), Label: "x",
			CanonicalURL: "https://example.test/a",
		}}
	case NodeImage:
		return NodeContent{Image: &ImageContent{URL: "https://cdn.example.test/a.png", Alt: "x"}}
	case NodeDateTime:
		return NodeContent{DateTime: when}
	case NodeMoney:
		return NodeContent{Money: &MoneyContent{DisplayValue: "1", Amount: "1", Currency: "USD"}}
	case NodeDeadline:
		return NodeContent{Deadline: when}
	case NodeOffer:
		return NodeContent{Offer: &OfferContent{Description: text, Terms: RichText{Runs: []InlineRun{}}}}
	case NodeLocation:
		return NodeContent{Location: &LocationContent{Name: "x", Address: PostalAddress{
			StreetAddress: "x", AddressLocality: "x", AddressCountry: "IN",
		}}}
	case NodeProduct:
		return NodeContent{Product: &ProductContent{Name: "x"}}
	case NodeFact:
		return NodeContent{Fact: &FactContent{Label: "x", Value: "1"}}
	case NodePersonalization:
		return NodeContent{Personalization: &PersonalizationContent{
			Slot: SlotReference{ID: "first_name", SemanticClass: SlotName, Fallback: "x"},
		}}
	case NodeStructuredData:
		return NodeContent{StructuredData: &StructuredDataContent{
			Enabled: false, Provider: ProviderGmail, RegistryVersion: GmailRegistryVersion,
			UseCase: UseCaseGmailPromotion, Kind: StructuredDiscountOffer,
			Declaration: StructuredDeclaration{DiscountOffer: &DiscountOfferDeclaration{OfferNodeID: "n055"}},
		}}
	case NodeDivider:
		return NodeContent{Divider: &DividerContent{}}
	case NodePoll:
		return NodeContent{Poll: &PollContent{
			PollID:   fmt.Sprintf("poll%03d", index),
			Question: text,
			Options: []PollOption{
				{OptionID: "a", Label: text},
				{OptionID: "b", Label: text},
			},
		}}
	}
	return NodeContent{}
}
